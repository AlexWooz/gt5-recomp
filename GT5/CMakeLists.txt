# The in-game settings menu (F1): Dear ImGui 1.93.0 (third_party/imgui, MIT, unmodified) on the D3D12 renderer,
# and on the Vulkan one when the build has it (gt5_menu_vk.cpp, imgui_impl_vulkan). Input, sizes and pages are
# shared (gt5_menu.cpp, gt5_menu_ui.cpp); each renderer has a file of its own for the drawing.
# The runtime calls it through runtime/settings/gt5_menu.h and gt5_menu_vk.h; a build without these sources
# links the runtime's empty stand-ins (gt5_menu_stub.c) instead.
set(GT5_IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
if(WIN32 AND EXISTS "${GT5_IMGUI_DIR}/imgui.cpp" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/menu/gt5_menu.cpp")
    set(GT5_MENU_SOURCES
        src/menu/gt5_menu.cpp
        src/menu/gt5_menu_ui.cpp
        src/menu/gt5_menu_dx12.cpp
        src/menu/gt5_binds.c)
    set(GT5_IMGUI_SOURCES
        "${GT5_IMGUI_DIR}/imgui.cpp" "${GT5_IMGUI_DIR}/imgui_draw.cpp" "${GT5_IMGUI_DIR}/imgui_tables.cpp"
        "${GT5_IMGUI_DIR}/imgui_widgets.cpp" "${GT5_IMGUI_DIR}/backends/imgui_impl_dx12.cpp")
    target_sources(gt5 PRIVATE ${GT5_MENU_SOURCES} ${GT5_IMGUI_SOURCES})
    set(_menu_inc "${GT5_IMGUI_DIR};${GT5_IMGUI_DIR}/backends;${PS3RECOMP_DIR}/runtime/settings")
    set(_menu_defs "IMGUI_DISABLE_DEMO_WINDOWS;IMGUI_DISABLE_DEBUG_TOOLS;IMGUI_DISABLE_OBSOLETE_FUNCTIONS")

    if(PS3RECOMP_RSX_VULKAN AND EXISTS "${GT5_IMGUI_DIR}/backends/imgui_impl_vulkan.cpp" AND PS3RECOMP_VULKAN_INCLUDE_DIR)
        set(GT5_MENU_VK_SOURCES src/menu/gt5_menu_vk.cpp "${GT5_IMGUI_DIR}/backends/imgui_impl_vulkan.cpp")

    else()
        message(STATUS "In-game menu:    on (Dear ImGui)")
    endif()
    target_link_libraries(gt5 PRIVATE d3dcompiler shell32)

# Scoped so the huge PPU TUs keep their cached compile commands. /GS-: no stack-cookie check in
# every lifted SPU function (most are a basic block long; their locals hold no guest-sized buffers).
if(SPU_SOURCES)
    set_source_files_properties(${SPU_SOURCES} PROPERTIES
        INCLUDE_DIRECTORIES "${PS3RECOMP_DIR}/runtime/spu"
        COMPILE_DEFINITIONS SPU_USE_DISPATCH_MT
        COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma;-ffp-contract=off;-DSPU_X86_RZ_ALWAYS=1")
endif()
# GT5's sound job: its numbers are samples, and it was about 10% short of real time with a full grid.
# jD58063CD is the disc's; game update 2.17 replaces it with jCB9BD024 (spu_jobs_from_eboot.py, UPDATED).
# Only the mixer: the downmix module (jFDFFAAC1) filled the mix with NaN under these rules (15 Nov),
# since the SPU reads exponent-255 words as large numbers where the host sees infinity or NaN.
foreach(_sound_job jD58063CD jCB9BD024)
    set_source_files_properties("${SPU_DIR}/jobs/${_sound_job}/spu_recomp.c" PROPERTIES
        COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma;-ffp-contract=off;-DSPU_X86_RZ_ALWAYS=1;-DSPU_X86_FAST_FLOAT=1")
endforeach()
# The EQ module (socepdideq: jEB79FDFC on the disc, jE3FDDA10 from 2.01) does its FFT's phase steps in
# double precision, three ops per pass, and switching the rounding mode around each one costs ten
# times the op: its doubles round toward zero instead (SPU_X86_FAST_DOUBLE, spu_helpers.h).
foreach(_eq_job jEB79FDFC jE3FDDA10)
    set_source_files_properties("${SPU_DIR}/jobs/${_eq_job}/spu_recomp.c" PROPERTIES
        COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma;-ffp-contract=off;-DSPU_X86_RZ_ALWAYS=1;-DSPU_X86_FAST_DOUBLE=1")
endforeach()

# Native replacements of hot SPU functions (src/native/natives.txt: image, address, symbol, source).
# The lifted job file is compiled from a patched copy in which the listed function is renamed and a
# switch with its old name picks native or lifted at run time (scripts/spu_native_patch.py), so no
# re-lift is needed and every caller is unchanged. GT5_SPU_NATIVE=0 runs the lifted code as before.
set(GT5_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/native")
# Direct branches (scripts/spu_native_patch.py --direct-bi, src/native/gt5_direct_bi.c): in the lifted code of
# every WWS job and of the WWS job manager, a branch or call to a register holding one of the image's own
# entries goes there without the runtime's dispatcher wherever the dispatcher would choose the same (the
# rest is unchanged). GT5_DIRECT_BI=0 at run time: as before; 2: each one checked against the dispatcher;
# GT5_DIRECT_BI_OFF=<image,...>: those images as before. Off here: none of it is built.
option(GT5_SPU_DIRECT_BI "direct branches in the lifted SPU job images" ON)
set(_dbi_args "")
set(_dbi_deps "")
if(GT5_SPU_DIRECT_BI AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/gen/spu_jobs.c")
    set(_dbi_args --direct-bi "${CMAKE_CURRENT_SOURCE_DIR}/src/gen/spu_jobs.c")
    set(_dbi_deps "${CMAKE_CURRENT_SOURCE_DIR}/src/gen/spu_jobs.c")
endif()
if(EXISTS "${GT5_NATIVE_DIR}/natives.txt" AND SPU_JOB_SOURCES)
    find_program(GT5_PYTHON NAMES python python3
                 HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../venv/Scripts" "${CMAKE_CURRENT_SOURCE_DIR}/../venv/bin")
    file(STRINGS "${GT5_NATIVE_DIR}/natives.txt" _nat_lines REGEX "^[A-Za-z0-9]")
    # Three-field lines say something about an image and are read first:
    #   <image> lift-sha1 <hex>   its native functions were written against exactly this lifted file;
    #                               with any other (another release, another lifter) the image stays lifted
    #   <image> generate <script>   src/native/<script> <lifted file> <build folder> writes files the
    #                               native source includes (statements cut from this copy's own lift)
    set(_nat_skip "")
    foreach(_l ${_nat_lines})
        separate_arguments(_f UNIX_COMMAND "${_l}")
        list(LENGTH _f _fn)
        if(NOT _fn EQUAL 3)
            continue()
        endif()
        list(GET _f 0 _img)
        list(GET _f 1 _kind)
        list(GET _f 2 _val)
        set(_in "${SPU_DIR}/jobs/${_img}/spu_recomp.c")
        if(NOT EXISTS "${_in}")
            continue()
        endif()
        if(_kind STREQUAL "lift-sha1")
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_in}")
            file(SHA1 "${_in}" _sha)
            if(NOT _sha STREQUAL _val AND NOT _img IN_LIST _nat_skip)
                list(APPEND _nat_skip ${_img})
                message(STATUS "Native SPU:      ${_img} stays lifted (its lifted code is not the one the native functions were written for)")
            endif()
        elseif(_kind STREQUAL "generate")
            list(APPEND _nat_gen_${_img} "${_val}")
        endif()
    endforeach()
    # -DGT5_CULL_PROF=ON (testing builds only): the w018CAD00 job profiler -- src/native/natives_prof.txt's
    # hooks and the profiler code in native_w018CAD00.c (GT5_CULL_PROF=1 at run time prints "[cull-prof]").
    # Off (the default, every player build): none of its hooks is in the game.
    option(GT5_CULL_PROF "w018CAD00 job profiler hooks (testing builds only)" OFF)
    set(_nat_lists "${GT5_NATIVE_DIR}/natives.txt")
    if(GT5_CULL_PROF AND EXISTS "${GT5_NATIVE_DIR}/natives_prof.txt")
        list(APPEND _nat_lists "${GT5_NATIVE_DIR}/natives_prof.txt")
    endif()
    # -DGT5_NATIVE_STATS=ON (testing): the geometry natives count what they do and print it every 5 s.
    set(_nat_stats "")
    if(GT5_NATIVE_STATS)
        set(_nat_stats ";NATIVE_STATS")
    endif()
    set(_nat_any OFF)
    set(_nat_done "")
    foreach(_l ${_nat_lines})
        separate_arguments(_f UNIX_COMMAND "${_l}")
        list(LENGTH _f _fn)
        if(_fn LESS 4)
            continue()
        endif()
        list(GET _f 0 _img)
        list(GET _f 3 _src)
        set(_in "${SPU_DIR}/jobs/${_img}/spu_recomp.c")
        if(NOT EXISTS "${_in}" OR NOT GT5_PYTHON OR _img IN_LIST _nat_skip)
            continue()   # this copy of the game has no such job (another release): nothing to replace
        endif()
        get_source_file_property(_opts "${_in}" COMPILE_OPTIONS)
        set(_incs "${PS3RECOMP_DIR}/runtime/spu;${SPU_DIR}/jobs/${_img};${GT5_NATIVE_DIR};${CMAKE_BINARY_DIR}/spu_native/${_img}")
        set(_stamp "")
        if(DEFINED _nat_gen_${_img})
            set(_stamp "${CMAKE_BINARY_DIR}/spu_native/${_img}/generated.stamp")
        endif()
        if(_stamp AND NOT _img IN_LIST _nat_done)
            set(_gen_cmds "")
            set(_gen_deps "${_in}")
            foreach(_g ${_nat_gen_${_img}})
                list(APPEND _gen_cmds COMMAND "${GT5_PYTHON}" "${GT5_NATIVE_DIR}/${_g}" "${_in}" "${CMAKE_BINARY_DIR}/spu_native/${_img}")
                list(APPEND _gen_deps "${GT5_NATIVE_DIR}/${_g}")
            endforeach()
            add_custom_command(OUTPUT "${_stamp}"
                ${_gen_cmds}
                COMMAND "${CMAKE_COMMAND}" -E touch "${_stamp}"
                DEPENDS ${_gen_deps}
                COMMENT "Native SPU: statements of ${_img} for its native functions" VERBATIM)
            target_sources(gt5 PRIVATE "${_stamp}")
        endif()
        if(NOT _img IN_LIST _nat_done)
            list(APPEND _nat_done ${_img})
            set(_out "${CMAKE_BINARY_DIR}/spu_native/${_img}/spu_recomp.c")
            add_custom_command(OUTPUT "${_out}"
                COMMAND "${GT5_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/spu_native_patch.py" "${_in}" "${_out}" ${_img} ${_nat_lists} ${_dbi_args}
                DEPENDS "${_in}" ${_nat_lists} ${_dbi_deps} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/spu_native_patch.py"
                COMMENT "Native SPU functions for ${_img}" VERBATIM)
            file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/spu_native/${_img}")
            set_source_files_properties("${_in}" PROPERTIES HEADER_FILE_ONLY ON)
            target_sources(gt5 PRIVATE "${_out}")
            set_source_files_properties("${_out}" PROPERTIES
                INCLUDE_DIRECTORIES "${_incs}" COMPILE_DEFINITIONS SPU_USE_DISPATCH_MT COMPILE_OPTIONS "${_opts}")
        endif()
        get_source_file_property(_have "${GT5_NATIVE_DIR}/${_src}" COMPILE_OPTIONS)
        if(NOT _have)
            target_sources(gt5 PRIVATE "${GT5_NATIVE_DIR}/${_src}")
            set(_nat_defs "SPU_USE_DISPATCH_MT;NATIVE_HOOKED")
            if(_src MATCHES "^native_w")   # the geometry natives; the sound ones keep their counters for the harness
                string(APPEND _nat_defs "${_nat_stats}")
            endif()
            if(GT5_CULL_PROF AND _src STREQUAL "native_w018CAD00.c")
                string(APPEND _nat_defs ";GT5_CULL_PROF_HOOKS")
            endif()
            set_source_files_properties("${GT5_NATIVE_DIR}/${_src}" PROPERTIES
                INCLUDE_DIRECTORIES "${_incs}" COMPILE_DEFINITIONS "${_nat_defs}" COMPILE_OPTIONS "${_opts}")
            if(_stamp)
                set_source_files_properties("${GT5_NATIVE_DIR}/${_src}" PROPERTIES OBJECT_DEPENDS "${_stamp}")
            endif()
        endif()
        set(_nat_any ON)
    endforeach()
    if(_nat_any)
        target_sources(gt5 PRIVATE "${GT5_NATIVE_DIR}/spu_native.c")
        set_source_files_properties("${GT5_NATIVE_DIR}/spu_native.c" PROPERTIES
            INCLUDE_DIRECTORIES "${PS3RECOMP_DIR}/runtime/spu"
            COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma")
        if(EXISTS "${GT5_NATIVE_DIR}/native_snd.c")
            target_sources(gt5 PRIVATE "${GT5_NATIVE_DIR}/native_snd.c")
            set_source_files_properties("${GT5_NATIVE_DIR}/native_snd.c" PROPERTIES
                INCLUDE_DIRECTORIES "${PS3RECOMP_DIR}/runtime/spu" COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma")
        endif()
        if(EXISTS "${GT5_NATIVE_DIR}/native_geo_switch.c")
            target_sources(gt5 PRIVATE "${GT5_NATIVE_DIR}/native_geo_switch.c")
            set_source_files_properties("${GT5_NATIVE_DIR}/native_geo_switch.c" PROPERTIES
                INCLUDE_DIRECTORIES "${PS3RECOMP_DIR}/runtime/spu;${GT5_NATIVE_DIR}" COMPILE_DEFINITIONS "NATIVE_GEO_SWITCH${_nat_stats}" COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma")
        endif()
        list(LENGTH _nat_done _nat_n)
        message(STATUS "Native SPU:      ${_nat_n} image(s) with native functions")
    endif()
endif()
if(SPU_JOB_SOURCES AND EXISTS "${GT5_NATIVE_DIR}/gt5_direct_bi.c")
    # the run-time half (also what the natives' in-run A/B calls), with or without the option
    target_sources(gt5 PRIVATE "${GT5_NATIVE_DIR}/gt5_direct_bi.c")
    set_source_files_properties("${GT5_NATIVE_DIR}/gt5_direct_bi.c" PROPERTIES
        INCLUDE_DIRECTORIES "${PS3RECOMP_DIR}/runtime/spu;${GT5_NATIVE_DIR}" COMPILE_OPTIONS "-O2;/w;/GS-;-mavx2;-mfma")
endif()
if(_dbi_args AND SPU_JOB_SOURCES)
    find_program(GT5_PYTHON NAMES python python3
                 HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../venv/Scripts" "${CMAKE_CURRENT_SOURCE_DIR}/../venv/bin")
    if(GT5_PYTHON)
        # The images with native functions got theirs in the natives patch above; every other job image is
        # compiled from a copy made here (the script copies an image that takes no direct branches unchanged).
        set(_dbi_n 0)
        foreach(_in ${SPU_JOB_SOURCES})
            get_filename_component(_dir "${_in}" DIRECTORY)
            get_filename_component(_img "${_dir}" NAME)
            if(_img IN_LIST _nat_done)
                continue()
            endif()
            set(_out "${CMAKE_BINARY_DIR}/spu_dbi/${_img}/spu_recomp.c")
            add_custom_command(OUTPUT "${_out}"
                COMMAND "${GT5_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/spu_native_patch.py" "${_in}" "${_out}" ${_img} ${_dbi_args}
                DEPENDS "${_in}" ${_dbi_deps} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/spu_native_patch.py"
                COMMENT "Direct branches for ${_img}" VERBATIM)
            file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/spu_dbi/${_img}")
            get_source_file_property(_opts "${_in}" COMPILE_OPTIONS)
            set_source_files_properties("${_in}" PROPERTIES HEADER_FILE_ONLY ON)
            target_sources(gt5 PRIVATE "${_out}")
            set_source_files_properties("${_out}" PROPERTIES
                INCLUDE_DIRECTORIES "${PS3RECOMP_DIR}/runtime/spu;${_dir};${GT5_NATIVE_DIR}"
                COMPILE_DEFINITIONS SPU_USE_DISPATCH_MT COMPILE_OPTIONS "${_opts}")
            math(EXPR _dbi_n "${_dbi_n} + 1")
        endforeach()
        message(STATUS "Direct branches: on (the WWS jobs' code and job manager; ${_dbi_n} job file(s) copied for it)")
    endif()
endif()

    # picks the Vulkan backend on it when RSX_LIVE_DRAW is off.
    set_property(SOURCE "${PS3RECOMP_DIR}/runtime/ppu/tests/boot_main.cpp" APPEND PROPERTY COMPILE_DEFINITIONS PS3RECOMP_RSX_VULKAN)
    # SDL2 loads only when Vulkan runs, so DirectX 12 starts even without SDL2.dll.
    target_link_options(gt5 PRIVATE /DELAYLOAD:SDL2.dll)
    target_link_libraries(gt5 PRIVATE delayimp)
    find_file(GT5_SDL2_DLL SDL2.dll PATH_SUFFIXES bin REQUIRED)
    add_custom_command(TARGET gt5 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GT5_SDL2_DLL}" "$<TARGET_FILE_DIR:gt5>")
else()
    # SDL2.dll next to gt5.exe tells the launcher and play.bat the exe has Vulkan: not this one.
    add_custom_command(TARGET gt5 POST_BUILD COMMAND ${CMAKE_COMMAND} -E rm -f "$<TARGET_FILE_DIR:gt5>/SDL2.dll")
endif()
set_target_properties(gt5 PROPERTIES LINKER_LANGUAGE CXX)

target_compile_options(gt5 PRIVATE /W3 /wd4100 /wd4244 /wd4267 /Zc:__cplusplus -fno-strict-aliasing -fwrapv)
target_compile_definitions(gt5 PRIVATE _CRT_SECURE_NO_WARNINGS)

# Machine-generated: warnings off, and /bigobj because one TU passes the default
# object format's section limit. -O1 keeps these TUs' build time and memory sane;
# the last -O on the line wins, so this overrides Release's /O2 for them alone.
# -O2 with this CPU's instructions (2026-11-05): the largest chunk builds in 71 s against 67 s at -O1, under 1 GB.
# MOVBE folds the byte swap of every big-endian guest load and store into the access itself.
# No FMA contraction, so floating-point results match the -O1 build. -mfma lets ppu_vmx.h do
# vmaddfp/vnmsubfp four lanes at once with the same results (every AVX2 CPU has FMA; contraction
# stays off, so nothing else changes).
set_source_files_properties(${RECOMP_SOURCES} PROPERTIES COMPILE_OPTIONS "/w;/bigobj;-O2;-mavx2;-mfma;-mbmi;-mbmi2;-mmovbe;-mlzcnt;-mpopcnt;-ffp-contract=off")

# Keep symbols so crash RVAs can be symbolized.
target_compile_options(gt5 PRIVATE /Z7)
target_link_options(gt5 PRIVATE /DEBUG /INCREMENTAL:NO)
